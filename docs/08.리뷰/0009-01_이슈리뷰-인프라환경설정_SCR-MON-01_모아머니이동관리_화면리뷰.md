## 2. 코드 레벨 핵심 분석

### 2.1 `vite.config.ts` 호스트 파싱 및 설정
```typescript
// 환경변수 ALLOWED_HOSTS가 있으면 쉼표 기준으로 자르고, 없으면 기본값 사용
const allowedHosts = process.env.ALLOWED_HOSTS
  ? process.env.ALLOWED_HOSTS.split(',').map((host) => host.trim())
  : ['dev.moabogo.com', 'stg.moabogo.com', 'www.moabogo.com', 'moabogo.com', 'localhost'];

export default defineConfig({
  plugins: [react(), tailwindcss()],
  server: {
    port: 3000,
    allowedHosts: allowedHosts,
  },
});
```

### 2.2 `server.ts` Vite Middleware 연동
```typescript
if (process.env.NODE_ENV !== 'production') {
  const allowedHosts = process.env.ALLOWED_HOSTS
    ? process.env.ALLOWED_HOSTS.split(',').map((host) => host.trim())
    : ['dev.moabogo.com', 'stg.moabogo.com', 'www.moabogo.com', 'moabogo.com', 'localhost'];

  const vite = await createViteServer({
    server: {
      middlewareMode: true,
      allowedHosts: allowedHosts,
    },
    appType: 'spa',
  });
  app.use(vite.middlewares);
}
```
